> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One generic unit component (`UNITS`) with a single table feeding every classifier, converter, formatter, and Guard B — canonical metric/Celsius base, unit-carrying import, safety-directional display conversion, and (Phase 10) full imperial display.

**Architecture:** A single `UNITS` namespace section in app.js built around a strict-JSON `UNITS_TABLE` between comment markers; build.py extracts the SAME table to generate Guard B's `_GB_UNIT_CLASS`. The five duplicate classifiers become derivations/wrappers; `UNITS.toCanonical` is the one import door, `UNITS.fmt` the one exit door. Spec: `docs/superpowers/specs/2026-07-31-units-architecture-design.md`.

**Tech Stack:** Vanilla JS (single-file PWA, `build.py` inlines), Python build guard, Playwright tests via `tests/_fixtures` (warm page).

**Phases:** Task 1–3 = **Phase 1** (U-1..U-3, ride with R-2/R-3) · Task 4–5 = **Phase 3** (U-4/U-5, precondition for CP3/Phase 7) · Task 6 = **Phase 10** (U-6, the switch + migration waves).

## Global Constraints

- **Safety invariance (DoD-10):** no `bcheck` stage, `temp`, `safe` value, or cook duration is altered — this plan never writes to `data.py` values; U-4 adds `cited` fields in `sources.py` ONLY (new keys, no value edits).
- **Dictionaries never convert** (Guard B invariant): no numeric change ever enters a `lang-*.json`; conversion is render/speak-time only, after the dict layer.
- **Cited values are never re-derived**; safety floors round only UP on display conversion (`role:'safeFloor'`), safety ceilings only DOWN (`'safeCeil'`).
- **Tests:** every test uses `test`/`seedApp` from `./_fixtures` (never raw Playwright `test`), waits on conditions never `waitForTimeout`, asserts observable effects (per `tests/TEST-AUTHORING-CONTRACT.md`). RED must be witnessed before GREEN.
- **Suite gate:** `npx playwright test` plain — no `--retries`, no `--workers`; exit code captured directly (`; ec=$?`), never through a pipe; no manual server on 8123; never run concurrently with heavy agents.
- **Build:** after any app.js/build.py change, `python build.py` before any UI check; restart any manual `serve.js` (it caches dist in memory).
- **Byte-identity rule:** while the display target is canonical (metric), `UNITS.fmt` output is verbatim `String(v)` + canonical label — every migrated site must render the identical string it rendered before.
- **Numeric readouts in RTL prose** get `dir="ltr"` islands (lesson L13) — `UNITS.fmtHtml` provides this.
- **The Task-13 seam is consumed, not bypassed:** `vcNormalizeSafetyText` delegates to `UNITS.normalize`; no second NFKC/degree-symbol path may appear.

// separate bare-"deg" behavior; the fragment list reproduces the exact original via composition below.
const UNITS=(function(){
  const T=UNITS_TABLE;
  function normalize(s){ return String(s||'').normalize('NFKC'); }
  function _key(u){ return normalize(u).toLowerCase().replace(/\s+/g,''); }
  // token → {kind, unit} exact-match map (whitespace-stripped, NFKC, lowercased — the 1b248a1 lesson
  // "strip ALL whitespace before classifying" holds by construction for every classifier at once).
  const _CLASS={};
  Object.keys(T.kinds).forEach(function(kind){
    const ku=T.kinds[kind].units;
    Object.keys(ku).forEach(function(unit){
      (ku[unit].tokens||[]).forEach(function(t){ _CLASS[_key(t)]={kind:kind, unit:unit}; });
    });
  });
  function classify(u){ const s=_key(u); if(!s) return null; return _CLASS[s]||null; }
  function _fwd(c){ return function(v){ return (v+c.pre)*c.num/c.den; }; }
  function legacyConv(){ const o={}; Object.keys(T.conv).forEach(function(k){ o[k]=_fwd(T.conv[k]); }); return o; }
  function toCanonical(value, unit, kind){
    if(value==null || isNaN(value)) return null;
    const k=T.kinds[kind]; if(!k) return null;
    const c=classify(unit); if(!c || c.kind!==kind) return null;
    const src={v:value, unit:String(unit)};
    if(c.unit==='' || c.unit===k.canon) return {v:value, unit:k.canon, src:src};   // canonical / scale-unspecified → passthrough
    const convKey=k.units[c.unit] && k.units[c.unit].conv;
    const cv=convKey && T.conv[convKey]; if(!cv) return null;                       // fail-closed
    return {v:_fwd(cv)(value), unit:k.canon, src:src};
  }
  function tokenizerUnitSrc(){ return '(?:'+[
    T.tokenizerFragments[0], T.tokenizerFragments[1], T.tokenizerFragments[2], T.tokenizerFragments[3],
    T.tokenizerFragments[4],
    // recompose the legacy deg-branch EXACTLY: deg(?:rees?)? — optional plural/singular tail —
    'deg(?:rees?)?(?:[^\\S\\r\\n]*(?:C\\b|F\\b|celsius\\b|fahrenheit\\b)|\\.?(?![A-Za-z]))',
    T.tokenizerFragments[6], T.tokenizerFragments[7]
  ].join('|')+')'; }
  function _reEscU(s){ return String(s).replace(/[.*+?^${}()|[\]\\]/g,'\\$&'); }
  function _prefixRe(tokens){
    const alts=tokens.slice().sort(function(a,b){ return b.length-a.length; }).map(function(t){
      const esc=_reEscU(t);
      return /^[a-z]+$/i.test(t) ? esc+'\\b' : esc;      // bare Latin words need a boundary; glyph/Hebrew forms do not
    });
    return new RegExp('^(?:'+alts.join('|')+')','i');
  }
  function vcClasses(){
    return T.vcOrder.map(function(row){
      let toks=[];
      row.kinds.forEach(function(kind){
        const ku=T.kinds[kind].units;
        Object.keys(ku).forEach(function(unit){ toks=toks.concat(ku[unit].tokens||[]); });
      });
      return {re:_prefixRe(toks), cls:row.cls};
    });
  }
  return {TABLE:T, KINDS:T.kinds, normalize:normalize, classify:classify, toCanonical:toCanonical,
          tokenizerUnitSrc:tokenizerUnitSrc, vcClasses:vcClasses, legacyConv:legacyConv, _key:_key};
})();
