# Voice Wave 0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Chunked, race-free, instrumented voice pipeline speaking the app's UI language, with visible errors, an instant acknowledgement, the HE/EN buttons removed, and a fail-closed Phonikud-shaped diacritization seam + niqqud A/B trial.

**Architecture:** All edits live inside the existing voice area of `app.js` (`vc*`/`gem*` family, ~lines 6411–6996 + the registry at ~5440-5480) — no module extraction (that is Phase 12). A pure chunker + a monotonic speaker-generation token feed a synthesize-ahead/play-in-order pipeline; diacritization is a post-guard, per-chunk, fail-closed hook.

**Tech Stack:** Vanilla JS single-file PWA (`build.py` inlines `app.js`), Playwright tests via `tests/_fixtures` (warm page, `seedApp`), Gemini TTS (`gemFetch('tts', …)`).

**Spec:** `docs/superpowers/specs/2026-07-31-voice-wave0-design.md` — read it first; every task cites its section.

- **Owner rulings (ROADMAP §5a):** Google TTS only (no premium tier) · Phonikud/diacritization is **Hebrew-only** · voice language = UI language (R-31) · HE/EN buttons removed (R-29).

```js
// ── Voice Wave 0 · sentence chunker (spec §2.1). ONE rule for all 7 live languages: split at
// sentence-ending punctuation FOLLOWED BY WHITESPACE — so "63.5" can never split. Short chunks merge
// forward (min); an overlong sentence hard-splits at the last comma/space before max — keeping every
// chunk inside the 20s TTS timeout and Chrome's ~15s single-utterance cutoff.
function vcChunkText(text, opts){
  opts=opts||{}; const MINC=opts.min||25, MAXC=opts.max||220;
  const s=String(text||'').replace(/\s+/g,' ').trim(); if(!s) return [];
  const parts=s.split(/(?<=[.!?…])\s+/);
  const merged=[];
  for(const p of parts){
    const prev=merged[merged.length-1];
    if(prev!==undefined && (prev.length<MINC || p.length<MINC) && (prev.length+p.length+1)<=MAXC)
      merged[merged.length-1]=prev+' '+p;
    else merged.push(p);
  }
  const out=[];
  for(let c of merged){
    while(c.length>MAXC){
      let cut=c.lastIndexOf(',', MAXC); if(cut<MINC) cut=c.lastIndexOf(' ', MAXC); if(cut<MINC) cut=MAXC;
      out.push(c.slice(0, cut+1).trim()); c=c.slice(cut+1).trim();
    }
    if(c) out.push(c);
  }
  return out;
}
```

```js
// prefetched WHILE chunk i plays (lookahead 1). First sound after ONE short synthesis, not the whole answer.
async function gemSynthChunk(clean){
  const key=clean+gemVoice();
  if(gemCache.has(key)) return gemCache.get(key);
  if(typeof window!=='undefined' && window.__gemTtsMock){ const b=window.__gemTtsMock(clean); gemCache.set(key,b); return b; }
  const r=await gemFetch('tts', {contents:[{parts:[{text:clean}]}], generationConfig: gemTtsGen(gemVoice())}, {timeout:20000});
  if(!r.ok){ let reason=''; try{const eb=await r.json(); reason=(eb.error&&eb.error.message)||'';}catch(_){}
    console.warn('Gemini TTS error', r.status, reason); throw new Error('api-'+r.status+(reason?': '+reason:'')); }
  const buf=gemReadAudio(await r.json()).buf;
  if(gemCache.size>40) gemCache.clear();
  gemCache.set(key, buf);
  return buf;

function gemPlayBuf(buf, gen){
  if(typeof window!=='undefined' && window.__gemPlayMock) return window.__gemPlayMock(buf, gen);
  return new Promise(function(res){
    gemCtx=gemCtx||new (window.AudioContext||window.webkitAudioContext)();
    const go=function(){ if(!vcGenCurrent(gen)){ res(); return; }
      gemSrc=gemCtx.createBufferSource(); gemSrc.buffer=buf; gemSrc.connect(gemCtx.destination);
      vcSpeaking=true; gemSrc.onended=function(){ vcSpeaking=false; res(); }; gemSrc.start(); };
    (gemCtx.state==='suspended') ? gemCtx.resume().then(go, go) : go();
  });
}
async function gemSpeak(text, lang, gen){

  const chunks=vcChunkText(speechText(text, lang||vcAnsLang()));
```

```js
// ── Voice Wave 0 · chunk-chained system voice (spec §2.4). One utterance per chunk, chained on
// onend/onerror — no single utterance can hit Chrome's ~15s cutoff, and a mid-chain stop is generation-aware.
function sysSpeakChunked(chunks, lang, gen){
  try{
    if(gen===undefined) gen=vcNewSpeakGen();
    speechSynthesis.cancel();
    let i=0;
    const step=function(){
      if(!vcGenCurrent(gen) || i>=chunks.length) return;
      const u=new SpeechSynthesisUtterance(chunks[i++]);
      u.lang=vcLocale(lang); u.rate=0.92; u.pitch=1;
```
